function doGet(e) { var p = e.parameter; var action = p.action; var passId = p.id; var devId = p.device; var props = PropertiesService.getScriptProperties(); // ૧. તમામ પાસનું લાઈવ એન્ટ્રી સ્ટેટસ શીટ માટે મોકલવું if (action === 'get_all_statuses') { var allProps = props.getProperties(); return ContentService.createTextOutput(JSON.stringify(allProps)) .setMimeType(ContentService.MimeType.JSON); } // ૨. સિક્યોરિટી ચેક: પહેલાં ફોન તપાસો જેથી ફોરવર્ડ થયેલ પાસ સીધો બ્લોક થાય if (action === 'check_lock') { var savedDev = props.getProperty('dev_' + passId); if (savedDev === 'CANCELLED_BY_ADMIN') { return ContentService.createTextOutput(JSON.stringify({ status: 'CANCELLED' })).setMimeType(ContentService.MimeType.JSON); } // પહેલી વાર ઓપન થયો - આ ફોન સાથે લૉક કરો if (!savedDev) { props.setProperty('dev_' + passId, devId); return ContentService.createTextOutput(JSON.stringify({ status: 'OK' })).setMimeType(ContentService.MimeType.JSON); } // જો આ એ જ અસલી ફોન હોય: if (savedDev === devId) { var isEntered = props.getProperty('entered_' + passId); if (isEntered === 'YES') { return ContentService.createTextOutput(JSON.stringify({ status: 'ALREADY_ENTERED' })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'OK' })).setMimeType(ContentService.MimeType.JSON); } // જો બીજા કોઈ ફોનમાં ફોરવર્ડ થયો હોય - સીધો બ્લોક! return ContentService.createTextOutput(JSON.stringify({ status: 'BLOCKED' })).setMimeType(ContentService.MimeType.JSON); } // ૩. ગેટ એન્ટ્રી પિન (1234) if (action === 'gate_entry') { if (p.pin === '1234') { props.setProperty('entered_' + passId, 'YES'); return ContentService.createTextOutput(JSON.stringify({ status: 'SUCCESS' })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'WRONG_PIN' })).setMimeType(ContentService.MimeType.JSON); } // ૪. પાસ રીસેટ if (action === 'reset') { props.deleteProperty('dev_' + passId); props.deleteProperty('entered_' + passId); return ContentService.createTextOutput(JSON.stringify({ status: 'RESET' })).setMimeType(ContentService.MimeType.JSON); } }
